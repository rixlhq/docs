    import {Video} from '@rixl/media-react';

    function VideoWithProgress({videoId}) {
      return (
        <Video
          id={videoId}
          className="w-full h-auto rounded-lg"
          controls
          analytics
          analyticsPage="feed"
          onRixlAnalytics={(e) => console.log(e.detail.event)}
        />
      );
    }
    ```

  </Tab>

  <Tab value="Image Gallery">
    ```jsx
    import {Image} from '@rixl/media-react';

    function ImageGallery({images}) {
      return (
        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-4">
          {images.map((image) => (
            <div key={image.id} className="aspect-square">
              <Image
                id={image.id}
                className="w-full h-full object-cover rounded-lg"
                analytics
                analyticsPage="standalone"
              />
            </div>
          ))}
        </div>
      );
    }
    ```

  </Tab>

  <Tab value="Content Feed">
    ```jsx

    function ContentFeed({feedId}) {
      return (
        <div className="max-w-2xl mx-auto">
          <Feed
            feedId={feedId}
            className="w-full space-y-6"
            loop
            autoscroll
            analytics
            analyticsPage="feed"
            onRixlAnalytics={(e) => console.log(e.detail.event)}
          />
        </div>
      );
    }
    ```

  </Tab>
</Tabs>

<Callout type="info">
  The React SDK wraps the Lit web components (`<rixl-video>`, `<rixl-image>`, `<rixl-feed>`) with `@lit/react`. Use `onRixlAnalytics` to receive analytics events. `Video` forwards standard media attributes (`autoPlay`, `muted`, `loop`, `playsInline`, `poster`, `volume`) and player UI props (`controls`, `theme`, `progressBar`, `showChapters`, `chapters`, `heatmap`, `hotSegments`, `allowFullscreen`, `allowPictureInPicture`, `autoHideMs`, `resumeProgress`, `soundDisabled`).
</Callout>

### API Integration Examples

<Tabs items={['Upload Video', 'Upload Image', 'Get Feed Content']}>
  <Tab value="Upload Video">
    ```javascript
    const uploadVideo = async (projectId, file, poster) => {
      // Reuse the same key if you retry this call, so it can't create a second upload.
      const idempotencyKey = crypto.randomUUID();
      const init = await fetch(`https://api.rixl.com/media/v1/projects/${projectId}/videos/upload`, {
        method: "POST",
        headers: {
          "X-API-Key": process.env.RIXL_API_KEY,
          "Content-Type": "application/json",
          "Idempotency-Key": idempotencyKey,
        },
        body: JSON.stringify({name: file.name, video_quality: "VIDEO_QUALITY_PRO"}),
      });
      if (!init.ok) throw new Error(`upload init failed: ${init.status}`);
      const {video_id, video_upload_url, poster_upload_url} = await init.json();

      await Promise.all([
        fetch(video_upload_url, {method: "PUT", body: file}),
        fetch(poster_upload_url, {method: "PUT", body: poster}),
      ]);
      return video_id;
    };
    ```

  </Tab>

  <Tab value="Upload Image">
    ```javascript
    const uploadImage = async (projectId, imageFile) => {
      const init = await fetch(`https://api.rixl.com/media/v1/projects/${projectId}/images/upload`, {
        method: "POST",
        headers: {"X-API-Key": process.env.RIXL_API_KEY, "Content-Type": "application/json"},
        body: JSON.stringify({name: imageFile.name}),
      });
      if (!init.ok) throw new Error(`upload init failed: ${init.status}`);
      const {image_id, upload_url} = await init.json();

      // Processing starts once the upload lands; there is no completion call.
      await fetch(upload_url, {method: "PUT", body: imageFile});
      return image_id;
    };
    ```

  </Tab>

  <Tab value="Get Feed Content">
    ```javascript
    // Public feeds need no API key.
    const getFeedContent = async (feedId, creatorId) => {
      const path = creatorId ? `posts/v1/feeds/${feedId}/creators/${creatorId}` : `posts/v1/feeds/${feedId}`;
      const response = await fetch(`https://api.rixl.com/${path}?pagination.limit=25`);
      if (!response.ok) throw new Error(`feed request failed: ${response.status}`);
      const {posts, total} = await response.json();
      return {posts, total};
    };
    ```
  </Tab>
</Tabs>
