## Implementation Examples

### React Component

```jsx
import {Image} from "@rixl/media-react";

export function ProfileHeader({imageId}) {
  return <Image id={imageId} className="w-full h-64 object-cover rounded-xl" alt="Profile banner" />;
}
```

### Direct API Ingestion

```javascript
// Step 1: Initialize image upload
const initResponse = await fetch(`https://api.rixl.com/media/v1/projects/${projectId}/images/upload`, {
  method: "POST",
  headers: {
    "X-API-Key": apiKey,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({name: imageFile.name}),
});

const {image_id, image_upload_url} = await initResponse.json();

// Step 2: Upload raw image binary
await fetch(image_upload_url, {
  method: "PUT",
  headers: {"Content-Type": imageFile.type || "image/jpeg"},
  body: imageFile,
});

console.log("Image uploaded successfully:", image_id);
```
