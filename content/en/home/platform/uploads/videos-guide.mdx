## Processing Pipeline Details

<Steps>
  <Step>
    ### Content Analysis
    **Automatic metadata extraction:**
    - Resolution, frame rate, and codec detection
    - Audio track analysis and separation
    - Content complexity assessment for bitrate selection
    - Aspect ratio preservation planning
  </Step>

  <Step>
    ### Intelligent Encoding
    **Multi-rendition transcoding:**
    - **Primary codec**: H.265 (HEVC) for efficiency
    - **Other codec**: H.264 for universal compatibility
    - **Frame rate handling**: 24fps to 60fps with NTSC support
    - **Adaptive bitrates**: Content-aware quality selection
  </Step>

  <Step>
    ### Adaptive Streaming
    **Cross-platform delivery preparation:**
    - **HLS packaging**: Standard `.m3u8` playlists compatible with modern mobile and web players
    - **Optimized segments**: Segmented for fast time-to-first-frame and smooth bitrate adaptation
    - **CDN distribution**: Global edge deployment for fast, low-latency streaming
  </Step>
</Steps>

## Processing Status Monitoring

Monitor your video's progress through the processing pipeline:

<Tabs items={['Status Types', 'API Monitoring']}>
  <Tab value="Status Types">
    - **`FILE_STATUS_UPLOADING`** – Upload initialized, awaiting video bytes
    - **`FILE_STATUS_UPLOADED`** – Bytes received, queued for transcoding
    - **`FILE_STATUS_PROCESSING`** – Video is being transcoded and packaged into HLS
    - **`FILE_STATUS_READY`** – Processing complete, ready for streaming
    - **`FILE_STATUS_ERROR`** – Processing failed
  </Tab>
  <Tab value="API Monitoring">
    ```bash
    # Check video details and processing status
    curl -X GET "https://api.rixl.com/media/v1/projects/{project_id}/videos/{video_id}" \
      -H "X-API-Key: YOUR_PROJECT_API_KEY"
    ```

    **Response includes:**
    - Current status (`file.status`)
    - Streaming format (`file.format`, flips from `"mp4"` to `"hls"` when ready)
    - File details (`duration`, `width`, `height`, `codec`, `bitrate`, `framerate`)

    ```typescript
    /**
     * Video upload with TypeScript and comprehensive error handling
     */

    interface CreateVideoUploadResponse {
      video_id: string;
      video_upload_url: string;
      poster_id?: string;
      poster_upload_url?: string;
      expires_at?: string;
    }

    async function uploadVideo(
      projectId: string,
      apiKey: string,
      videoFile: File,
      videoQuality: 'VIDEO_QUALITY_BASIC' | 'VIDEO_QUALITY_SHORTS' | 'VIDEO_QUALITY_PRO' = 'VIDEO_QUALITY_PRO'
    ): Promise<string> {
      try {
        // Step 1: Create upload URL
        const initResponse = await fetch(
          `https://api.rixl.com/media/v1/projects/${projectId}/videos/upload`,
          {
            method: 'POST',
            headers: {
              'X-API-Key': apiKey,
              'Content-Type': 'application/json',
              'Accept': 'application/json'
            },
            body: JSON.stringify({
              name: videoFile.name,
              video_quality: videoQuality
            })
          }
        );

        if (!initResponse.ok) {
          const errorData = await initResponse.json().catch(() => ({}));
          throw new Error(`Upload initialization failed: ${errorData.message || initResponse.statusText}`);
        }

        const uploadData: CreateVideoUploadResponse = await initResponse.json();

        // Step 2: Upload video file directly to signed URL
        const videoUpload = await fetch(uploadData.video_upload_url, {
          method: 'PUT',
          body: videoFile,
          headers: {
            'Content-Type': videoFile.type || 'video/mp4'
          }
        });

        if (!videoUpload.ok) {
          throw new Error(`Video upload failed: ${videoUpload.statusText}`);
        }

        return uploadData.video_id;
      } catch (error) {
        console.error('Upload failed:', error);
        throw error;
      }
    }

    // Usage example
    const handleUpload = async (file: File) => {
      const videoId = await uploadVideo('YOUR_PROJECT_ID', 'YOUR_PROJECT_API_KEY', file, 'VIDEO_QUALITY_PRO');
      console.log(`Upload successful! Video ID: ${videoId}`);
    };
    ```

    ```python
    """
    Python video upload with error handling and type hints
    """

    import requests
    from pathlib import Path
    import logging

    logging.basicConfig(level=logging.INFO)
    logger = logging.getLogger(__name__)

    class VideoUploadError(Exception):
        """Custom exception for video upload errors"""
        pass

    def upload_video(
        project_id: str,
        api_key: str,
        file_path: Path,
        video_quality: str = "VIDEO_QUALITY_PRO"
    ) -> str:
        """
        Upload a video file to RIXL platform.

        Args:
            project_id: RIXL project ID
            api_key: Your RIXL project API key
            file_path: Path to the video file
            video_quality: Quality tier ('VIDEO_QUALITY_BASIC', 'VIDEO_QUALITY_SHORTS', 'VIDEO_QUALITY_PRO')

        Returns:
            str: Video ID of the uploaded video
        """
        if not file_path.exists():
            raise VideoUploadError(f"Video file not found: {file_path}")

        session = requests.Session()
        session.headers.update({
            "X-API-Key": api_key,
            "User-Agent": "RIXL-Python-Client/1.0"
        })

        try:
            # Step 1: Initialize upload
            logger.info(f"Initializing upload for {file_path.name}")
            init_response = session.post(
                f"https://api.rixl.com/media/v1/projects/{project_id}/videos/upload",
                json={
                    "name": file_path.name,
                    "video_quality": video_quality
                },
                timeout=30
            )
            init_response.raise_for_status()
            upload_data = init_response.json()

            # Step 2: Upload video file directly to signed URL
            logger.info("Uploading video file...")
            with open(file_path, "rb") as video_file:
                upload_response = session.put(
                    upload_data["video_upload_url"],
                    data=video_file,
                    headers={"Content-Type": "video/mp4"},
                    timeout=300
                )
                upload_response.raise_for_status()

            video_id = upload_data["video_id"]
            logger.info(f"Upload successful! Video ID: {video_id}")
            return video_id

        except requests.exceptions.RequestException as e:
            raise VideoUploadError(f"Upload request failed: {e}") from e

    # Usage example
    if __name__ == "__main__":
        try:
            video_id = upload_video(
                project_id="YOUR_PROJECT_ID",
                api_key="YOUR_PROJECT_API_KEY",
                file_path=Path("my-video.mp4"),
                video_quality="VIDEO_QUALITY_PRO"
            )
            print(f"Success! Video ID: {video_id}")
        except VideoUploadError as e:
            logger.error(f"Upload failed: {e}")
    ```
