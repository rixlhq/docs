RIXL provides framework-free web components via `@rixl/media-lit` and headless state management via `@rixl/media`. Vue 3 has first-class native support for custom elements, enabling you to use `<rixl-video>`, `<rixl-image>`, and `<rixl-feed>` directly in any Vue or Nuxt application with standard reactive props and event listeners.

## Installation

Install `@rixl/media-lit` alongside the core `@rixl/media` package:

```bash
npm install @rixl/media-lit @rixl/media
```

## Vite Configuration

Configure Vue to treat `rixl-*` tags as custom elements so the template compiler does not emit unknown element warnings:

```ts title="vite.config.ts"
import vue from "@vitejs/plugin-vue";
import {defineConfig} from "vite";

export default defineConfig({
  plugins: [
    vue({
      template: {
        compilerOptions: {
          isCustomElement: (tag) => tag.startsWith("rixl-"),
        },
      },
    }),
  ],
});
```

<Callout type="info">
  If you are using **Nuxt 3**, configure `vue.compilerOptions.isCustomElement` in your `nuxt.config.ts` under the `vue` section.
</Callout>

## Authentication

Initialize the RIXL client once in your application root or entry file before rendering media components:

```ts
import {connect} from "@rixl/media";

await connect({
  baseUrl: "https://api.rixl.com",
  apiKey: "your-api-key",
});
```

## Video Player

Import the `@rixl/media-lit/video` element and bind your video ID dynamically:

```vue title="VideoPlayer.vue"
<script setup lang="ts">
import "@rixl/media-lit/video";

defineProps<{
  videoId: string;
}>();
</script>

<template>
  <div class="video-container">
    <rixl-video :video-id="videoId" autoplay controls />
  </div>
</template>
```

## Responsive Images

Import `@rixl/media-lit/image` to render optimized, responsive images with automatic WebP/AVIF format negotiation:

```vue title="ImageViewer.vue"
<script setup lang="ts">
import "@rixl/media-lit/image";

defineProps<{
  imageId: string;
}>();
</script>

<template>
  <rixl-image :image-id="imageId" alt="Media asset" />
</template>
```

## Media Feeds

Display infinite-scroll feeds or story carousels using `<rixl-feed>`:

```vue title="MediaFeed.vue"
<script setup lang="ts">
import "@rixl/media-lit/feed";

defineProps<{
  feedId: string;
}>();
</script>

<template>
  <rixl-feed :feed-id="feedId" loop autoscroll />
</template>
```

## Events & Analytics

The media components dispatch native custom events that can be listened to with Vue's `@` syntax:

```vue title="AnalyticsExample.vue"
<script setup lang="ts">
import "@rixl/media-lit/video";

function handleAnalytics(event: CustomEvent) {
  console.log("Playback event:", event.detail);
}
</script>

<template>
  <rixl-video video-id="vid_123" @rixl-analytics="handleAnalytics" />
</template>
```
